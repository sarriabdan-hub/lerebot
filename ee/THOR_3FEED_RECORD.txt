WHY
  The workstation records the replay by polling Thor's /observe over HTTP, so it
  only grabs a handful of frames per second -> choppy, frame-skipping video.
  Thor already HAS every frame at the control-loop rate. This patch makes Thor
  record a single side|top|wrist composite clip PER RUN, locally, at full rate,
  and serve it over HTTP so the workstation can pull it for the replay UI.

  Result: the 3-feed videos are (a) saved locally on Thor in ee/thor_3feed/, and
  (b) shown in the workstation dashboard's Replay tab at full fps.

*** VLA PERFORMANCE IS NOT AFFECTED ***
  The control loop does essentially no extra work: per step it only COPIES the 3
  small camera arrays (~0.3 ms memcpy) and drops them on a queue. ALL the heavy
  work — resize, compositing, and video encoding — runs on a SEPARATE background
  thread. If that thread ever falls behind, frames are DROPPED, never blocked. So
  the 30 Hz control rate (and thus the arm's motion) is unchanged. You can prove
  it: compare `control_hz_mean` in /state with vs without --record-3feed — same.

# ── 3-feed composite recorder (all cams, full rate, OFF the control loop) ───────
# Records side|top|wrist as ONE mp4 per execute, locally on Thor (ee/thor_3feed/).
# PERFORMANCE: the control loop only COPIES the 3 camera arrays (~0.3 ms) and hands
# them to the background encoder thread below; ALL resize/composite/encode work runs
# on that thread. If it falls behind, frames are DROPPED (never blocks the loop), so
# the VLA control rate is untouched. The workstation pulls the clip via GET /last_clip
# (+ /last_clip_info for timing) and retimes it to real-time for the replay.
REC_TILE_H = 360                                  # per-tile height (drop to 288 for even less CPU)
REC_CAMS = ("cam_side", "cam_top", "cam_wrist")   # left-to-right order in the composite
REC_QCAP = 300                                    # max queued frames; beyond this we drop (encoder keeps up)
_REC = {"path": None, "q": None, "stop": None, "thread": None, "n": 0, "t0": 0.0, "active": False}


def _rec_composite(item: dict):
    """side | top | wrist -> one BGR frame (black tile for a missing cam so the size stays constant)."""
    tiles, any_real = [], False
    for cam in REC_CAMS:
        rgb = item.get(cam)
        if rgb is None:
            tiles.append(np.zeros((REC_TILE_H, int(REC_TILE_H * 4 / 3), 3), np.uint8))
        else:
            bgr = cv2.cvtColor(rgb, cv2.COLOR_RGB2BGR)
            h, w = bgr.shape[:2]
            tiles.append(cv2.resize(bgr, (int(w * REC_TILE_H / h), REC_TILE_H)))
            any_real = True
    return np.hstack(tiles) if any_real else None


def _rec_encoder():
    """Background thread: composite + encode queued frames. All cv2 work lives here, OFF the control loop."""
    dq, stop = _REC["q"], _REC["stop"]
    vw, size, n = None, None, 0
    while True:
        try:
            item = dq.popleft()                   # deque.popleft is atomic in CPython (no lock needed)
        except IndexError:
            if stop.is_set():
                break
            time.sleep(0.003)
            continue
        frame = _rec_composite(item)
        if frame is None:
            continue
        if vw is None:
            h, w = frame.shape[:2]
            size = (w, h)
            vw = cv2.VideoWriter(str(_REC["path"]), cv2.VideoWriter_fourcc(*"mp4v"), float(FPS), (w, h))
        if (frame.shape[1], frame.shape[0]) != size:
            frame = cv2.resize(frame, size)
        vw.write(frame)
        n += 1
    if vw is not None:
        vw.release()
    _REC["n"] = n                                 # publish the final frame count for the timing meta


def _rec_start():
    """Begin a new 3-feed recording session + its background encoder thread."""
    if not STATE.get("record_3feed"):
        return
    _rec_finish()                                 # close any stale session first
    d = Path(__file__).parent / "thor_3feed"
    d.mkdir(parents=True, exist_ok=True)
    _REC["path"] = d / f"{int(time.time())}.mp4"
    _REC["q"] = deque()
    _REC["stop"] = threading.Event()
    _REC["n"] = 0
    _REC["t0"] = time.perf_counter()
    _REC["thread"] = threading.Thread(target=_rec_encoder, name="rec-encoder", daemon=True)
    _REC["active"] = True
    _REC["thread"].start()


def _rec_write(obs_raw: dict):
    """Control-loop side: COPY the 3 cam arrays (cheap memcpy) and hand off. Never blocks; drops if full."""
    if not _REC["active"]:
        return
    item = {}
    for cam in REC_CAMS:
        rgb = _find_cam_frame(obs_raw, cam)
        if rgb is not None:
            item[cam] = rgb.copy()                # copy so the encoder can't race the next get_observation
    if not item:
        return
    dq = _REC["q"]
    if dq is not None and len(dq) < REC_QCAP:     # drop instead of blocking if the encoder is behind
        dq.append(item)


def _rec_finish():
    """Finalize: stop the encoder, let it drain the queue, and record timing for /last_clip_info."""
    if not _REC["active"]:
        return
    _REC["active"] = False
    elapsed = max(time.perf_counter() - _REC["t0"], 0.001)   # capture window = t0..now (drain doesn't count)
    if _REC["stop"] is not None:
        _REC["stop"].set()
    if _REC["thread"] is not None:
        _REC["thread"].join(timeout=15.0)         # wait for the queued frames to finish encoding
    n = _REC["n"]
    STATE["last_clip"] = str(_REC["path"])
    STATE["last_clip_meta"] = {"n_frames": n, "elapsed": round(elapsed, 3),
                               "nominal_fps": float(FPS), "real_fps": round(n / elapsed, 2)}
    print(f"[3feed] {_REC['path']}  ({n} frames, {elapsed:.1f}s, {n / elapsed:.1f} fps real)")
    _REC["q"] = None
    _REC["thread"] = None
    _REC["stop"] = None
